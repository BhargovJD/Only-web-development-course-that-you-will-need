1. Create table
CREATE TABLE chai_store (
    id BIGSERIAL PRIMARY KEY,
    chai_name VARCHAR(255) NOT NULL,
    price DECIMAL(10, 2) NOT NULL,
    chai_type VARCHAR(100) NOT NULL,
    available BOOLEAN NOT NULL DEFAULT TRUE
);

3. Show

4. Specific 
SELECT chai_name AS "Chai Name", price AS "Price in INR" FROM public.chai_store

5. Search
SELECT * FROM public.chai_store WHERE chai_name = 'Green Tea'
ORDER BY id ASC 

6. Like
SELECT * FROM public.chai_store WHERE  chai_name LIKE '%Tea%'
ORDER BY id ASC 

7. Condition
SELECT * FROM public.chai_store WHERE  price < 30;

8. ORDER BY
SELECT * FROM chai_store  ORDER BY price DESC;
